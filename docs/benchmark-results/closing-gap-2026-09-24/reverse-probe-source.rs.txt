//! Private format-feasibility probe; not a production index writer.
use fst::{IntoStreamer, Streamer};
use hermes_core::directories::{Directory, MmapDirectory};
use hermes_core::structures::{AsyncSSTableReader, SSTableValue, TermInfo};
use std::{io::{BufWriter, Write}, path::Path, time::Instant};

fn digest(values:&[TermInfo])->(usize,u64,u64) {
    let mut bytes=vec![];
    for v in values {v.serialize(&mut bytes).unwrap()}
    // Stable checksum for diagnostics, not a file-integrity mechanism.
    (values.len(),values.iter().map(|v|v.doc_freq() as u64).sum(),bytes.iter().fold(0xcbf29ce484222325u64,|a,&b|(a^b as u64).wrapping_mul(0x100000001b3)))
}
#[tokio::main(flavor="current_thread")]
async fn main()->Result<(),Box<dyn std::error::Error>> {
    let args:Vec<_>=std::env::args().collect();let index=Path::new(&args[1]);let out=Path::new(&args[2]);std::fs::create_dir_all(out)?;
    let directory=MmapDirectory::new(index);
    let name=std::fs::read_dir(index)?.map(|p|p.unwrap().path()).find(|p|p.extension().is_some_and(|s|s=="terms")).unwrap();
    let reader=AsyncSSTableReader::<TermInfo>::open_with_cache_budget(directory.open_read(Path::new(name.file_name().unwrap())).await?,1024,Some(16*1024*1024)).await?;
    let original_path=Path::new(&args[3]);
    let fst_path=out.join("reverse.fst");
    if !fst_path.exists() {
        use std::io::{BufRead, BufReader};
        let start=Instant::now();
        let original_file=std::fs::File::open(original_path)?;
        let original=fst::Map::new(unsafe { memmap2::Mmap::map(&original_file)? })?;
        let unsorted=out.join("reverse.unsorted"); let sorted=out.join("reverse.sorted");
        let mut file=BufWriter::new(std::fs::File::create(&unsorted)?);
        let mut stream=original.range().ge([0u8;4]).lt([0,0,0,1]).into_stream();
        let mut count=0_u64;
        while let Some((key,_))=stream.next() {
            if key.len()<4 || key[..4]!=[0;4] || std::str::from_utf8(&key[4..]).is_err() {continue}
            for byte in key[4..].iter().rev() {write!(file,"{byte:02x}")?;}
            writeln!(file)?; count+=1;
        }
        file.flush()?; drop(file); drop(stream); drop(original);
        let status=std::process::Command::new("sort").env("LC_ALL","C")
            .args(["-S", "256M", "--parallel=1", "-T"]).arg(out)
            .arg("-o").arg(&sorted).arg(&unsorted).status()?;
        assert!(status.success());
        let mut builder=fst::MapBuilder::new(BufWriter::new(std::fs::File::create(&fst_path)?))?;
        for line in BufReader::new(std::fs::File::open(&sorted)?).lines() {
            let line=line?; let key:Vec<u8>=line.as_bytes().chunks_exact(2).map(|pair| u8::from_str_radix(std::str::from_utf8(pair).unwrap(),16).unwrap()).collect();
            builder.insert(key,0)?;
        }
        let mut file=builder.into_inner()?;file.flush()?;drop(file);
        println!("{}",serde_json::json!({"build_terms":count,"build_ms":start.elapsed().as_secs_f64()*1000.0,"fst_bytes":std::fs::metadata(&fst_path)?.len(),"sort_scratch_limit_bytes":268435456,"hex_sort_input_bytes":std::fs::metadata(&unsorted)?.len()}));
        std::fs::remove_file(unsorted)?; std::fs::remove_file(sorted)?;
    }
    let fst_file=std::fs::File::open(&fst_path)?;let mmap=unsafe{memmap2::Mmap::map(&fst_file)?};let map=fst::Map::new(mmap)?;
    for query in ["g*itar","h*band","q*lity","ht*p","wh*h","ne*w","*band"] {
        let (prefix,suffix)=query.split_once('*').unwrap();let mut keyprefix=0_u32.to_le_bytes().to_vec();keyprefix.extend_from_slice(prefix.as_bytes());
        let reversed_suffix: Vec<u8> = suffix.as_bytes().iter().rev().copied().collect();
        let scan=||reader.prototype_pattern_values(&keyprefix,|key|key.len()>=keyprefix.len()+suffix.len() && key.ends_with(suffix.as_bytes()) && std::str::from_utf8(&key[4..]).is_ok()).unwrap();
        let matched=|| {
            if prefix.len() >= suffix.len() { return scan(); }
            // Exact byte-prefix range on a reversed-term dictionary. Canonical
            // values are fetched from the existing SSTable only after matching.
            let mut upper=reversed_suffix.clone();
            let has_upper=if let Some(i)=upper.iter().rposition(|&b|b!=255){upper[i]+=1;upper.truncate(i+1);true}else{false};
            let range=map.range().ge(&reversed_suffix);
            let range=if has_upper {range.lt(&upper)} else {range};
            let mut stream=range.into_stream(); let mut rows=vec![];
            while let Some((reversed,_))=stream.next() {
                let term:Vec<u8>=reversed.iter().rev().copied().collect();
                if term.len()<prefix.len()+suffix.len() || !term.starts_with(prefix.as_bytes()) {continue}
                let mut key=0_u32.to_le_bytes().to_vec();key.extend_from_slice(&term);
                rows.push((key.clone(),reader.get_sync(&key).unwrap().unwrap()));
            }
            rows.sort_unstable_by(|a,b|a.0.cmp(&b.0));
            rows.into_iter().map(|(_,value)|value).collect::<Vec<_>>()
        };
        let expected=scan();let actual=matched();assert_eq!(actual,expected,"{query}");
        let mut rows=vec![];
        for label in ["scan1","fst1","fst2","scan2"] {
            let mut samples=vec![];
            for i in 0..40 {
                let start=Instant::now();let values=if label.starts_with("scan"){scan()}else{matched()};let elapsed=start.elapsed().as_nanos() as u64;
                assert_eq!(digest(&values),digest(&expected));if i>=5 {samples.push(elapsed)}
            }
            rows.push(serde_json::json!({"label":label,"samples_ns":samples}));
        }
        println!("{}",serde_json::json!({"query":query,"terms":expected.len(),"doc_freq_sum":digest(&expected).1,"timings":rows}));
    }
    Ok(())
}
